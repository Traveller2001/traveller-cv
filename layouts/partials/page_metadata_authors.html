"equal" }}{{ $mark = "*" }}{{ else if in (lower .) "corresponding" }}{{ $mark = "†" }}{{ end }}{{ end -}} {{- if .RelPermalink -}}{{ $name }}{{- else -}}{{ $name }}{{- end -}} {{ with $mark }}{{ . }}{{ end }} {{- end -}} {{- end -}}
